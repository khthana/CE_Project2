class BetterSocket
{
	Socket			theSocket;
	BufferedReader	theReader;
	PrintWriter		theWriter;

	public BetterSocket (String host,int port)
		throws UnknownHostException,IOException
	{
			this(new Socket(host,port));
	}

	public BetterSocket(Socket s)
		throws UnknownHostException,IOException
	{
		theSocket = s;
		theReader = new BufferedReader(
			new InputStreamReader(
				theSocket.getInputStream()));
		theWriter = new PrintWriter(
			theSocket.getOutputStream(),true);
	}

	public Socket getSocket()
		{
			return theSocket;
		}

	public void close()
		throws IOException
		{
			theSocket.close();
		}
				
	public String readLine()
		throws IOException
		{
			return theReader.readLine();
		}

	public void println(String s)
		throws IOException
		{
			theWriter.println(s);
		}
}

class MyItemListener implements ItemListener{
		public void itemStateChanged(ItemEvent e){
				Genpic ap = new Genpic();
				Object ev = e.getItem();
				if (ev.toString() == "Line")
					{
						ap.model = "Line";
						ap.state = 0;
					}
				if (ev.toString() == "Square")
					{
						ap.model = "Square";
						ap.state = 0;
					}
				if (ev.toString() == "Triangle")
					{
						ap.model = "Triangle";
						ap.state = 0;
					}				
		}
}

class MyMouseListener implements MouseListener{
		Genpic ap = new Genpic();
		public void mousePressed(MouseEvent e){
			//System.out.println("mouse pressed");
			//if (ap.model == "Line"){	
			if (ap.state == 0){
				ap.refresh = 0;
				ap.state = 1;
				ap.x_init = e.getX();
				ap.y_init = e.getY();
			}
			if (ap.state == 2){
				ap.tri_state = 1;
				int [] x = {ap.x_init,ap.x_bet,ap.x_end,ap.x_init};
				int [] y = {ap.y_init,ap.y_bet,ap.y_end,ap.y_init};
				ap.ig.drawPolyline(x,y,4);
				ap.board.drawpic();
				ap.state = 0;
			}
			//}
			//pr("pressed",e);
		}
		public void mouseReleased(MouseEvent e){
			int startx;
			int starty;
			int endx;
			int endy;
			//System.out.println("mouse release");
			if (ap.state == 1){
					ap.x_end = e.getX();
					ap.y_end = e.getY();
			}
			if (ap.model == "Line"){
				if (ap.state == 1){
					ap.ig.drawLine(ap.x_init,ap.y_init,ap.x_end,ap.y_end);
					ap.board.drawpic();
					ap.state = 0;
				}
			}
			if (ap.model == "Square"){
				if (ap.state == 1){
					if (ap.x_init < ap.x_end)
						{
						startx = ap.x_init;
						endx = ap.x_end;
						}else 
							{
							startx = ap.x_end;
							endx = ap.x_init;
							}
					if (ap.y_init < ap.y_end)
						{
						starty = ap.y_init;
						endy = ap.y_end;
						}else 
							{
							starty = ap.y_end;
							endy = ap.y_init;
							}
					int width = Math.abs(startx - endx);
					int height = Math.abs(starty - endy);
					ap.ig.drawRect(startx,starty,width,height);
					ap.board.drawpic();
					ap.state = 0;
				}
			}
			if (ap.model == "Triangle"){
				if (ap.state == 1){
					ap.ig.drawLine(ap.x_init,ap.y_init,ap.x_end,ap.y_end);
					ap.board.drawpic();
					ap.x_bet = ap.x_end;
					ap.y_bet = ap.y_end;
					ap.state = 2;
				}
			}
			//pr("released",e);
			//System.out.println("x1 = " + ap.x_init + " y1 = " + ap.y_init + " x2 = " + ap.x_end + " y2 = " + ap.y_end);
		}
