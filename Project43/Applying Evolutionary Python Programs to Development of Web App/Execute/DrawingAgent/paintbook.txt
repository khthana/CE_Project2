    class recieve_handler(Threading.Thread):
        def addobj(self, obj1, obj2):
            self.client = obj1
            self.connect = obj2

        def run(self):
            global Quit
            while not Quit:
                time.sleep(1)
                __data_in = self.connect.recv(1024)
                print __data_in
                if not __data_in: break
                __c = -1
                __job_r1 = ''
                __job_r2 = ''
                __job_r3 = ''
                __job_list = split(__data_in[:-3], ':')
                print __job_list
                for __job in __job_list:
                    __c = __c + 1
                    if __c % 3 == 0:
                        __job_r1 = __job_r1 + ', ' + __job
                    elif __c % 3 == 1:
                        __job_r2 = __job_r2 + ', ' + __job
                    else:
                        __job_r3 = __job_r3 + ', ' + __job
                print __job_r1
                print __job_r2
                print __job_r3
                if __job_r1 != '':
                    robot1.sendcmd(__job_r1[2:])
                if __job_r2 != '':
                    robot2.sendcmd(__job_r2[2:])
                if __job_r3 != '':
                    robot3.sendcmd(__job_r3[2:])
            self.connect.close()
            self.client.conn_list.remove(self.connect)
            print 'Disconnected.'

    def __send_methods(self):
        __data_out = '#addmethod1:'
        for __m in Methods1:
            if __m[0:2] != '__':
                __data_out = __data_out + __m + '&'
        Conn.send(__data_out + '\r\n')
        __data_out = '#addmethod2:'
        for __m in Methods2:
            if __m[0:2] != '__':
                __data_out = __data_out + __m + '&'
        Conn.send(__data_out + '\r\n')
        __data_out = '#addmethod3:'
        for __m in Methods3:
            if __m[0:2] != '__':
                __data_out = __data_out + __m + '&'
        Conn.send(__data_out + '\r\n')

    def run(self):
        global Server_Socket
        global Conn
        global Quit
        self.conn_list = []
        Server_Socket = socket(AF_INET, SOCK_STREAM)
        Server_Socket.bind(self.Host, self.Port)
        Server_Socket.listen(1)
        print 'Listening for Connection...'
        Conn, addr = Server_Socket.accept()
        print 'Connected by', addr
        self.__send_methods()
        while not Quit:
            self.con, addr = Server_Socket.accept()
            self.conn_list.append(self.con)
            print 'Connected by', addr
            reciever = self.recieve_handler()
            reciever.start()
            reciever.addobj(self, self.con)
