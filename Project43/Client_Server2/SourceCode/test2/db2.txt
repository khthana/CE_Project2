create table customer (
	username varchar(50) primary key,
	password varchar(50) not null,
	name varchar(100) not null,
	no varchar(100) not null,
	soi varchar(100),
	street varchar(100) not null,
	district varchar(100),
	province varchar(100) not null,
	country varchar(100) not null,
	area_code varchar(100) not null,
	tel varchar(100),
	fax varchar(100),
	email varchar(100),
	credit_ID varchar(100) not null,
	user_type varchar(100)
);
create table product (
	product_id varchar(100) primary key,
	category varchar(100) not null,
	brand varchar(100),
	quantity number,
	last_receive_date date,
	import_price number,
	sell_price number,
	low_sell_quantity number,
	sell_now varchar(100)
);
create table personalCom (
	product_id varchar(100),
	type varchar(100),
	model varchar(100),
	bus varchar(100),
	interfac varchar(100),
	chipset varchar(100),
	speed varchar(100),
	capacity varchar(100),
	more varchar(255),
	foreign key (product_id) references product (product_id),
	primary key (product_id)
);
create table notebook (
	product_id varchar(100),
	cpu varchar(100),
	ram varchar(100),
	hdd varchar(100),
	cd_rom varchar(100),
	fdd varchar(100),
	display varchar(100),
	fax_modem varchar(100),
	more varchar(255),
	foreign key (product_id) references product (product_id),
	primary key (product_id)
);
create table buy (
	username varchar(100),
	product_id varchar(100),
	sell_quantity number,
	total_price number,
	total_import_price number,
	sell_date date,
	sell_time varchar(10),
	verify varchar(10),
	receipt_no varchar(100),
	foreign key (username) references customer (username),
	foreign key (product_id) references product (product_id),
	primary key (username,product_id,receipt_no)
);


create table attrib (
	category varchar(100),
	attribute varchar(30),
	use varchar(10),
	no varchar(2),
	primary key (category,attribute)
);

create table bank (
	credit_id varchar(30) primary key,
	password varchar(30),
	balance number
);

create table picture(
	product_id varchar(100) primary key,
	file_pic1 varchar(50),
	file_pic2 varchar(50),
	show varchar(10)
);
